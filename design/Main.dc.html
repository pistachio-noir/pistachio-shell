1use std::fs;

4/// Count the lines in a file, like `wc -l`.
5fn count_lines(path: &Path) -> std::io::Result<usize> {
6 let text = fs::read_to_string(path)?;
7 Ok(text.lines().count())
8}
10fn main() {

12 for name in &args[1..] {
13 match count_lines(Path::new(name)) {

15 Err(e) => eprintln!("wc: {name}: {e}"),
16 }
17 }
18}
